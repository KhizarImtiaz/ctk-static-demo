Overview & Cardinal Rules

What we're building: Ask CTK is an invisible AI layer that silently enriches claim fields after an adjuster saves. The adjuster clicks Save and gets an instant response. In the background, a worker polls a SQL queue every 30 seconds, calls Google Gemini via the existing ctk-mcp-nls Python service, and writes AI text back to designated DB columns.

Increment 1 scope: When an adjuster saves a party statement (Facts record), AI generates a factual summary and writes it to Facts.AiSummary. This establishes the complete pipeline; all other integration points follow the same pattern.

CARDINAL RULE: AI MUST NEVER run inside the user save. Save enqueues only. Background worker drains the queue. If enqueue fails, log and continue — do NOT block the save.

Existing patterns to follow exactly

  • Repository pattern: Constructor takes (IDbConnection conn, IDbTransaction? transaction = null)
  • UnitOfWork lazy loading: private IRepo? _repo; public IRepo Repo => _repo ??= new Repo(GetTenantConnection(), _tenantTx);
  • BeginAsync() resets: All tenant repo fields set to null in BeginAsync()
  • Tenant ID: Comes from X-Tenant-ID HTTP header. In background worker: must call tenantProvider.SetTenantId(companyId.ToString()) before any DB access
  • Services: All application services registered as AddScoped
  • SQL style: Dapper with C# raw string literals ("""..."""), parameterized with @ParamName

    Task WriteAuditAsync(long queueId, int companyId, int toolkitId, string eventType, string? message, string? metadataJson = null);

using System.Data;
using CTK.Common.DTOs.AskCtk;
using CTK.Core.Interfaces;
using Dapper;

namespace CTK.Infrastructure.Repositories;

public class AskCtkQueueRepository : IAskCtkQueueRepository
{
    private readonly IDbConnection _conn;
    private readonly IDbTransaction? _transaction;

    public AskCtkQueueRepository(IDbConnection conn, IDbTransaction? transaction = null)
    {
        _conn = conn;
        _transaction = transaction;
    }

    public async Task<long?> EnqueueAsync(EnqueueJobRequest request)
    {
        // INSERT only when no active job exists for this exact record+type combination.
        // This is the dedup guard — prevents duplicate AI work when adjuster saves quickly.
        const string sql = """
            INSERT INTO [dbo].[AutoAskCTKUpdateQueue]
                ([CompanyId],[ToolkitId],[SourceTableName],[SourceRecordId],
                 [AiFunctionType],[AiRequestType],[PromptTypeKey],[OutputFormat],
                 [Status],[NotBeforeAt])
            OUTPUT INSERTED.[QueueId]
            SELECT @CompanyId, @ToolkitId, @SourceTableName, @SourceRecordId,
                   @AiFunctionType, @AiRequestType, @PromptTypeKey, @OutputFormat,
                   'PENDING', @NotBeforeAt
            WHERE NOT EXISTS (
                SELECT 1 FROM [dbo].[AutoAskCTKUpdateQueue]
                WHERE [CompanyId]       = @CompanyId
                  AND [ToolkitId]       = @ToolkitId
                  AND [SourceTableName] = @SourceTableName
                  AND [SourceRecordId]  = @SourceRecordId
                  AND [AiRequestType]   = @AiRequestType
                  AND [Status] IN ('PENDING','PROCESSING','RETRY')
            )
            """;

        return await _conn.QueryFirstOrDefaultAsync<long?>(sql, new
        {
            request.CompanyId,
            request.ToolkitId,
            request.SourceTableName,
            request.SourceRecordId,
            request.AiFunctionType,
            request.AiRequestType,
            request.PromptTypeKey,
            request.OutputFormat,
            request.NotBeforeAt
        }, transaction: _transaction);
    }

    public async Task<QueuedJobDto?> ClaimNextAsync(string workerId)
    {
        // Two-step: SELECT with UPDLOCK to atomically claim the row, then UPDATE status.
        // READPAST skips locked rows so concurrent workers don't block each other.
        const string sql = """
            DECLARE @id BIGINT;

            SELECT TOP 1 @id = [QueueId]
            FROM [dbo].[AutoAskCTKUpdateQueue] WITH (UPDLOCK, READPAST, ROWLOCK)
            WHERE [Status] IN ('PENDING', 'RETRY')
              AND (
                  ([Status] = 'PENDING' AND ([NotBeforeAt] IS NULL OR [NotBeforeAt] <= SYSUTCDATETIME()))
                  OR ([Status] = 'RETRY' AND ([NextRetryAt] IS NULL OR [NextRetryAt] <= SYSUTCDATETIME()))
              )
            ORDER BY [Priority] ASC, [CreatedAt] ASC;

            IF @id IS NOT NULL
            BEGIN
                UPDATE [dbo].[AutoAskCTKUpdateQueue]
                SET [Status]              = 'PROCESSING',
                    [LockedBy]            = @WorkerId,
                    [LockedAt]            = SYSUTCDATETIME(),
                    [ProcessingStartedAt] = SYSUTCDATETIME(),
                    [UpdatedAt]           = SYSUTCDATETIME()
                WHERE [QueueId] = @id;
            END

            SELECT [QueueId],[CompanyId],[ToolkitId],[SourceTableName],[SourceRecordId],
                   [AiFunctionType],[AiRequestType],[PromptTypeKey],[OutputFormat],
                   [Status],[RetryCount],[MaxRetryCount],[NotBeforeAt],[CreatedAt]
            FROM [dbo].[AutoAskCTKUpdateQueue]
            WHERE [QueueId] = @id;
            """;

        return await _conn.QueryFirstOrDefaultAsync<QueuedJobDto>(sql,
            new { WorkerId = workerId }, transaction: _transaction);
    }

    public async Task CompleteAsync(long queueId, string responseText, int? inputTokens, int? outputTokens)
    {
        const string sql = """
            UPDATE [dbo].[AutoAskCTKUpdateQueue]
            SET [Status]                = 'COMPLETED',
                [AiResponseRaw]         = @ResponseText,
                [AiResponseNormalized]  = @ResponseText,
                [InputTokenCount]       = @InputTokens,
                [OutputTokenCount]      = @OutputTokens,
                [ProcessingCompletedAt] = SYSUTCDATETIME(),
                [UpdatedAt]             = SYSUTCDATETIME()
            WHERE [QueueId] = @QueueId
            """;

        await _conn.ExecuteAsync(sql, new
        {
            QueueId = queueId,
            ResponseText = responseText,
            InputTokens = inputTokens,
            OutputTokens = outputTokens
        }, transaction: _transaction);
    }

    public async Task FailAsync(long queueId, string errorCode, string errorMessage, bool isRetryable, int retryDelayMinutes)
    {
        // RetryCount is incremented first, then compared to MaxRetryCount.
        // If retryable and still under limit → RETRY with NextRetryAt set.
        // Otherwise → FAILED permanently.
        const string sql = """
            UPDATE [dbo].[AutoAskCTKUpdateQueue]
            SET [RetryCount]   = [RetryCount] + 1,
                [ErrorCode]    = @ErrorCode,
                [ErrorMessage] = @ErrorMessage,
                [Status]       = CASE
                    WHEN @IsRetryable = 1 AND ([RetryCount] + 1) < [MaxRetryCount] THEN 'RETRY'
                    ELSE 'FAILED'
                END,
                [NextRetryAt]  = CASE
                    WHEN @IsRetryable = 1 AND ([RetryCount] + 1) < [MaxRetryCount]
                    THEN DATEADD(MINUTE, @RetryDelayMinutes, SYSUTCDATETIME())
                    ELSE NULL
                END,
                [UpdatedAt]    = SYSUTCDATETIME()
            WHERE [QueueId] = @QueueId
            """;

        await _conn.ExecuteAsync(sql, new
        {
            QueueId = queueId,
            ErrorCode = errorCode,
            ErrorMessage = errorMessage,
            IsRetryable = isRetryable,
            RetryDelayMinutes = retryDelayMinutes
        }, transaction: _transaction);
    }

    public async Task WriteAuditAsync(long queueId, int companyId, int toolkitId,
        string eventType, string? message, string? metadataJson = null)
    {
        const string sql = """
            INSERT INTO [dbo].[AskCtk_AuditLog]
                ([QueueId],[CompanyId],[ToolkitId],[EventType],[Message],[MetadataJson])
            VALUES (@QueueId, @CompanyId, @ToolkitId, @EventType, @Message, @MetadataJson)
            """;

        await _conn.ExecuteAsync(sql, new
        {
            QueueId = queueId,
            CompanyId = companyId,
            ToolkitId = toolkitId,
            EventType = eventType,
            Message = message,
            MetadataJson = metadataJson
        }, transaction: _transaction);
    }
}

using CTK.Common.DTOs.AskCtk;
using CTK.Core.Interfaces;
using Microsoft.Extensions.Logging;

namespace CTK.Application.AskCtk;

public class AskCtkJobProcessor
{
    private readonly IUnitOfWork _uow;
    private readonly IPromptRepository _promptRepo;
    private readonly IClaimContextBuilder _contextBuilder;
    private readonly IAskCtkAiClient _aiClient;
    private readonly IAiResultWriterRegistry _writerRegistry;
    private readonly ILogger<AskCtkJobProcessor> _logger;

    public AskCtkJobProcessor(
        IUnitOfWork uow,
        IPromptRepository promptRepo,
        IClaimContextBuilder contextBuilder,
        IAskCtkAiClient aiClient,
        IAiResultWriterRegistry writerRegistry,
        ILogger<AskCtkJobProcessor> logger)
    {
        _uow            = uow;
        _promptRepo     = promptRepo;
        _contextBuilder = contextBuilder;
        _aiClient       = aiClient;
        _writerRegistry = writerRegistry;
        _logger         = logger;
    }

    public async Task ProcessJobAsync(QueuedJobDto job, CancellationToken ct)
    {
        _logger.LogInformation(
            "[AskCtk] Processing job {QueueId} type={AiRequestType} record={SourceRecordId}",
            job.QueueId, job.AiRequestType, job.SourceRecordId);

        try
        {
            // 1. Load prompt template from DB (uses existing PromptRepository.ResolveAsync)
            var prompt = await _promptRepo.ResolveAsync(job.PromptTypeKey, job.CompanyId);
            if (prompt == null)
                throw new InvalidOperationException(
                    $"Prompt not found for key '{job.PromptTypeKey}' (companyId={job.CompanyId})");

            // 2. Build context (fetches Facts record + party name)
            var ctx = await _contextBuilder.BuildStatementContextAsync(
                (int)job.SourceRecordId, job.ToolkitId, job.CompanyId);
            if (ctx == null)
                throw new InvalidOperationException(
                    $"Context not found for Facts {job.SourceRecordId} toolkit={job.ToolkitId}");

            // 3. Render prompt (replace {{placeholders}} with actual values)
            var renderedPrompt = RenderPrompt(prompt.PromptText, ctx);

            // 4. Call AI (Gemini via ctk-mcp-nls, or mock)
            var aiResp = await _aiClient.CompleteAsync(new AiCompleteRequest
            {
                SystemInstruction = "You are a senior auto liability claims analyst. Be concise and factual.",
                Prompt            = renderedPrompt,
                OutputFormat      = job.OutputFormat
            }, ct);

            if (string.IsNullOrWhiteSpace(aiResp.Text))
                throw new InvalidOperationException("AI returned an empty response.");

            // 5. Find the correct writer for this AiRequestType
            var writer = _writerRegistry.GetWriter(job.AiRequestType)
                ?? throw new InvalidOperationException(
                    $"No IAiResultWriter registered for AiRequestType '{job.AiRequestType}'");

            // 6. Write AI result to DB column
            await writer.WriteAsync(job, aiResp.Text, ct);

            // 7. Mark job COMPLETED + write audit row
            await _uow.AskCtkQueue.CompleteAsync(
                job.QueueId, aiResp.Text, aiResp.InputTokens, aiResp.OutputTokens);
            await _uow.AskCtkQueue.WriteAuditAsync(
                job.QueueId, job.CompanyId, job.ToolkitId, "COMPLETED",
                $"Wrote {aiResp.Text.Length} chars to {job.SourceTableName}[{job.SourceRecordId}]");

            _logger.LogInformation("[AskCtk] Job {QueueId} completed ({Chars} chars)",
                job.QueueId, aiResp.Text.Length);
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "[AskCtk] Job {QueueId} failed: {Message}",
                job.QueueId, ex.Message);

            // Non-retryable: config errors, missing records, empty prompt
            bool isRetryable = ex is not InvalidOperationException;
            int delayMinutes = job.RetryCount switch { 0 => 1, 1 => 5, _ => 15 };

            await _uow.AskCtkQueue.FailAsync(
                job.QueueId, ex.GetType().Name, ex.Message, isRetryable, delayMinutes);
            await _uow.AskCtkQueue.WriteAuditAsync(
                job.QueueId, job.CompanyId, job.ToolkitId, "FAILED", ex.Message);
        }
    }

    private static string RenderPrompt(string template, StatementContextDto ctx) =>
        template
            .Replace("{{party_name}}",      ctx.PartyName       ?? "Unknown")
            .Replace("{{fact_description}}", ctx.FactDescription ?? "N/A")
            .Replace("{{right_of_way}}",    ctx.RightofWay      ?? "N/A")
            .Replace("{{look_out}}",        ctx.LookOut         ?? "N/A")
            .Replace("{{avoidance}}",       ctx.Avoidance       ?? "N/A")
            .Replace("{{speed}}",           ctx.Speed           ?? "N/A")
            .Replace("{{point_of_impact}}", ctx.PointofImpact   ?? "N/A");
}
NEW CTK.API\HostedServices\ProcessAskCtkQueueService.cs

CRITICAL: SetTenantId MUST be called before any IUnitOfWork/DbContext access. There is no HTTP request in a background worker, so the tenant ID is read from appsettings.json → Tenant:CompanyId.

using CTK.Application.AskCtk;
using CTK.Application.Interfaces;
using CTK.Core.Interfaces;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;

namespace CTK.API.HostedServices;

public class ProcessAskCtkQueueService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly ILogger<ProcessAskCtkQueueService> _logger;
    private readonly TimeSpan _interval = TimeSpan.FromSeconds(30);

    // Unique worker ID — used to tag rows claimed by this instance
    private readonly string _workerId =
        $"worker-{Environment.MachineName}-{Guid.NewGuid():N}";

    public ProcessAskCtkQueueService(
        IServiceScopeFactory scopeFactory,
        ILogger<ProcessAskCtkQueueService> logger)
    {
        _scopeFactory = scopeFactory;
        _logger       = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("[AskCtk Worker] Started. WorkerId={WorkerId}", _workerId);

        while (!stoppingToken.IsCancellationRequested)
        {
            try
            {
                await ProcessNextJobAsync(stoppingToken);
            }
            catch (Exception ex) when (!stoppingToken.IsCancellationRequested)
            {
                _logger.LogError(ex, "[AskCtk Worker] Unhandled error in tick");
            }

            await Task.Delay(_interval, stoppingToken);
        }

        _logger.LogInformation("[AskCtk Worker] Stopped.");
    }

    private async Task ProcessNextJobAsync(CancellationToken ct)
    {
        using var scope = _scopeFactory.CreateScope();

        // ── CRITICAL: Set tenant before any DB access ──────────────────────
        // Each deployed replica serves exactly one company. CompanyId comes from
        // appsettings.json → Tenant:CompanyId (set per deployment).
        var config        = scope.ServiceProvider.GetRequiredService<IConfiguration>();
        var tenantProvider = scope.ServiceProvider.GetRequiredService<ITenantProvider>();
        var companyId     = config.GetValue<int>("Tenant:CompanyId");
        tenantProvider.SetTenantId(companyId.ToString());
        // ────────────────────────────────────────────────────────────────────

        var uow = scope.ServiceProvider.GetRequiredService<IUnitOfWork>();
        var job = await uow.AskCtkQueue.ClaimNextAsync(_workerId);

        if (job == null)
            return; // Queue empty — nothing to do this tick

        _logger.LogInformation(
            "[AskCtk Worker] Claimed job {QueueId} type={AiRequestType}",
            job.QueueId, job.AiRequestType);

        var processor = scope.ServiceProvider.GetRequiredService<AskCtkJobProcessor>();
        await processor.ProcessJobAsync(job, ct);
    }
}

function getRelativeTime(date: Date): string {
  const diffMs  = Date.now() - date.getTime();
  const diffMin = Math.floor(diffMs / 60_000);
  if (diffMin < 1)  return 'just now';
  if (diffMin < 60) return `${diffMin} min ago`;
  const diffHr = Math.floor(diffMin / 60);
  if (diffHr < 24)  return `${diffHr} hr ago`;
  return `${Math.floor(diffHr / 24)} days ago`;
}
